cmake_minimum_required(VERSION 3.20)
project(CrashHybrid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "CrashHybrid.exe requires Windows")
endif()
if(NOT CRASH_REBUILT_DIR OR NOT CRASH_ORIGINAL_DIR)
  message(FATAL_ERROR "Set -DCRASH_REBUILT_DIR and -DCRASH_ORIGINAL_DIR")
endif()
foreach(path "${CRASH_REBUILT_DIR}/src/game_win32.cpp" "${CRASH_REBUILT_DIR}/src/png.cpp"
             "${CRASH_ORIGINAL_DIR}/src/rom_window_win32.cpp" "${CRASH_ORIGINAL_DIR}/CMakeLists.txt")
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Missing required source: ${path}")
  endif()
endforeach()
add_subdirectory("${CRASH_ORIGINAL_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/rom-original")
set(REBUILT_ENTRY "${CRASH_REBUILT_DIR}/src/game_win32.cpp")
set(ORIGINAL_ENTRY "${CRASH_ORIGINAL_DIR}/src/rom_window_win32.cpp")
add_executable(CrashHybrid WIN32 hybrid_win32.cpp "${REBUILT_ENTRY}" "${CRASH_REBUILT_DIR}/src/png.cpp" "${ORIGINAL_ENTRY}")
set_source_files_properties("${REBUILT_ENTRY}" PROPERTIES COMPILE_DEFINITIONS "WinMain=CrashRebuiltWinMain")
set_source_files_properties("${ORIGINAL_ENTRY}" PROPERTIES COMPILE_DEFINITIONS "WinMain=CrashOriginalWinMain")
target_include_directories(CrashHybrid PRIVATE "${CRASH_REBUILT_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(CrashHybrid PRIVATE /utf-8 /EHsc)
target_link_libraries(CrashHybrid PRIVATE gba_runtime user32 gdi32 winmm)
